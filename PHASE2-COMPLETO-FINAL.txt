🔧 COMO USAR
═════════════════════════════════════════════════════════════════════════════════

1️⃣  DESENVOLVIMENTO LOCAL

# Instalar
npm install

# Iniciar servidor
npm run start:server

# Rodar testes Jest
npm test

# Rodar testes E2E
npm run test:e2e
npm run test:e2e:ui

2️⃣  DOCKER LOCAL

# Build da imagem
docker build -t ppp-poker:latest .

# Rodar com compose
docker-compose up -d

# Ver logs
docker-compose logs -f app

# Parar
docker-compose down

3️⃣  TESTES

# Testes unitários
npm test

# Testes de integração
npm test socketio.test.js

# Testes E2E
npm run test:e2e

# Com UI interativa
npm run test:e2e:ui
